cmake_minimum_required(VERSION 3.24)
project(LibreWheel VERSION 0.1.0 LANGUAGES CXX)

if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported build configurations" FORCE)
endif()

option(BUILD_TESTING "Build automated tests" ON)
if(BUILD_TESTING)
    enable_testing()
endif()

set(LIBREWHEEL_PGO_MODE "OFF" CACHE STRING "MSVC Release driver PGO mode (OFF, TRAIN, USE)")
set_property(CACHE LIBREWHEEL_PGO_MODE PROPERTY STRINGS OFF TRAIN USE)
set(LIBREWHEEL_PGO_PROFILE "" CACHE FILEPATH "Driver profile database (defaults to LibreWheelFF.pgd in the build directory)")
set(LIBREWHEEL_PAYLOAD_FILE "" CACHE FILEPATH "Packed LZMS payload embedded into LibreWheel-Setup.exe")

set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_DEBUG OFF)
if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

    # Release: full MSVC optimizations + cross-file IPO/LTCG, aggressive inlining, intrinsics, COMDAT folding.
    set(CMAKE_CXX_FLAGS_RELEASE "/O2 /Ob3 /Oi /Ot /Gw /Gy /DNDEBUG" CACHE STRING "" FORCE)
    set(CMAKE_EXE_LINKER_FLAGS_RELEASE "/LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO /DEBUG:NONE" CACHE STRING "" FORCE)
    set(CMAKE_SHARED_LINKER_FLAGS_RELEASE "/LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO /DEBUG:NONE" CACHE STRING "" FORCE)
    set(CMAKE_STATIC_LINKER_FLAGS_RELEASE "/LTCG" CACHE STRING "" FORCE)

    # Debug: zero optimizations, full debug symbols (.pdb) and runtime checks.
    set(CMAKE_CXX_FLAGS_DEBUG "/Od /Ob0 /Zi /RTC1 /D_DEBUG" CACHE STRING "" FORCE)
    set(CMAKE_EXE_LINKER_FLAGS_DEBUG "/DEBUG:FULL /OPT:NOREF /OPT:NOICF" CACHE STRING "" FORCE)
    set(CMAKE_SHARED_LINKER_FLAGS_DEBUG "/DEBUG:FULL /OPT:NOREF /OPT:NOICF" CACHE STRING "" FORCE)
endif()

if(NOT LIBREWHEEL_PGO_MODE STREQUAL "OFF")
    if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        message(FATAL_ERROR "LibreWheel PGO requires the native MSVC toolchain on Windows.")
    endif()
    if(NOT LIBREWHEEL_PGO_PROFILE)
        set(LIBREWHEEL_PGO_PROFILE "${CMAKE_CURRENT_BINARY_DIR}/LibreWheelFF.pgd")
    endif()
    if(LIBREWHEEL_PGO_MODE STREQUAL "TRAIN")
        set(pgo_option GENPROFILE)
    elseif(LIBREWHEEL_PGO_MODE STREQUAL "USE")
        if(NOT EXISTS "${LIBREWHEEL_PGO_PROFILE}")
            message(FATAL_ERROR "LibreWheel PGO profile was not found: ${LIBREWHEEL_PGO_PROFILE}")
        endif()
        set(pgo_option USEPROFILE)
    else()
        message(FATAL_ERROR "LIBREWHEEL_PGO_MODE must be OFF, TRAIN, or USE.")
    endif()
endif()

add_library(wheel_core STATIC src/effects.cpp src/settings.cpp src/protocol.cpp)
target_include_directories(wheel_core PUBLIC src)
target_compile_features(wheel_core PUBLIC cxx_std_20)

if(WIN32)
    function(librewheel_add_version_info target filetype filename description)
        set(LW_RC_FILETYPE "${filetype}")
        set(LW_RC_FILENAME "${filename}")
        set(LW_RC_DESCRIPTION "${description}")
        set(rc_file "${CMAKE_CURRENT_BINARY_DIR}/version_${target}.rc")
        configure_file(src/windows/version.rc.in "${rc_file}" @ONLY)
        target_sources(${target} PRIVATE "${rc_file}")
    endfunction()

    add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN DIRECTINPUT_VERSION=0x0800
                            _WIN32_WINNT=0x0A00)
    add_library(wheel_hid STATIC src/windows/hid.cpp)
    target_link_libraries(wheel_hid PUBLIC wheel_core hid setupapi)

    add_executable(LibreWheelOutput src/windows/broker.cpp)
    target_link_libraries(LibreWheelOutput PRIVATE wheel_hid)
    librewheel_add_version_info(LibreWheelOutput VFT_APP "LibreWheelOutput.exe" "LibreWheel G29 USB Output Broker")

    add_library(LibreWheelFF SHARED src/windows/driver.cpp src/windows/output_lease.cpp src/windows/driver.def)
    target_link_libraries(LibreWheelFF PRIVATE wheel_hid dxguid uuid)
    add_dependencies(LibreWheelFF LibreWheelOutput)
    librewheel_add_version_info(LibreWheelFF VFT_DLL "LibreWheelFF.dll" "LibreWheel DirectInput Force-Feedback Driver")
    if(MSVC)
        target_compile_options(LibreWheelFF PRIVATE /guard:cf)
        target_link_options(LibreWheelFF PRIVATE /GUARD:CF)
    endif()
    if(pgo_option)
        target_link_options(LibreWheelFF PRIVATE
            "$<$<CONFIG:Release>:/${pgo_option}:PGD=${LIBREWHEEL_PGO_PROFILE}>")
    endif()

    add_executable(librewheel src/windows/cli.cpp)
    target_link_libraries(librewheel PRIVATE wheel_hid dinput8 dxguid uuid cfgmgr32)
    librewheel_add_version_info(librewheel VFT_APP "librewheel.exe" "LibreWheel Diagnostic CLI")

    add_executable(LibreWheelTray WIN32 src/windows/tray.cpp src/windows/tray.rc src/windows/tray.manifest)
    target_link_libraries(LibreWheelTray PRIVATE wheel_hid shell32 user32 advapi32)
    librewheel_add_version_info(LibreWheelTray VFT_APP "LibreWheelTray.exe" "LibreWheel System Tray")

    add_executable(pack_payload src/windows/pack_payload.cpp)
    target_link_libraries(pack_payload PRIVATE wheel_core cabinet)

    if(LIBREWHEEL_PAYLOAD_FILE)
        if(NOT EXISTS "${LIBREWHEEL_PAYLOAD_FILE}")
            message(FATAL_ERROR "LibreWheel installer payload was not found: ${LIBREWHEEL_PAYLOAD_FILE}")
        endif()
        file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/windows/LibreWheel.ico" LW_INSTALLER_ICON)
        file(TO_CMAKE_PATH "${LIBREWHEEL_PAYLOAD_FILE}" LW_PAYLOAD_FILE)
        set(installer_rc "${CMAKE_CURRENT_BINARY_DIR}/installer.rc")
        configure_file(src/windows/installer.rc.in "${installer_rc}" @ONLY)
        set_source_files_properties("${installer_rc}" PROPERTIES OBJECT_DEPENDS "${LIBREWHEEL_PAYLOAD_FILE}")
        add_executable(LibreWheel-Setup WIN32
            src/windows/installer.cpp
            src/windows/registration.cpp
            "${installer_rc}"
            src/windows/tray.manifest)
        target_include_directories(LibreWheel-Setup PRIVATE src/windows)
        target_compile_definitions(LibreWheel-Setup PRIVATE "LIBREWHEEL_VERSION_STR=L\"${PROJECT_VERSION}\"")
        target_link_libraries(LibreWheel-Setup PRIVATE wheel_hid cabinet comctl32 dwmapi ole32 shell32 advapi32 user32 uuid)
        if(MSVC)
            target_compile_options(LibreWheel-Setup PRIVATE /guard:cf)
            target_link_options(LibreWheel-Setup PRIVATE /GUARD:CF)
        endif()
    endif()
endif()

if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE wheel_core)
    add_test(NAME core COMMAND core_tests)
    if(WIN32)
        add_executable(driver_tests tests/driver_tests.cpp src/windows/driver.cpp)
        target_link_libraries(driver_tests PRIVATE wheel_hid dxguid uuid)
        target_compile_definitions(driver_tests PRIVATE LIBREWHEEL_TEST_DRIVER)
        add_test(NAME driver COMMAND driver_tests $<TARGET_FILE:LibreWheelFF>)

        add_executable(broker_test_helper src/windows/broker.cpp)
        target_link_libraries(broker_test_helper PRIVATE wheel_core)
        target_compile_definitions(broker_test_helper PRIVATE LIBREWHEEL_TEST_BROKER)

        add_executable(output_tests tests/output_tests.cpp src/windows/output_lease.cpp)
        target_link_libraries(output_tests PRIVATE wheel_hid)
        add_test(NAME output COMMAND output_tests $<TARGET_FILE:broker_test_helper>)

        add_executable(installer_tests tests/installer_tests.cpp src/windows/registration.cpp)
        target_link_libraries(installer_tests PRIVATE wheel_hid cabinet advapi32)
        add_test(NAME installer COMMAND installer_tests $<TARGET_FILE:LibreWheelFF>)
    endif()
endif()
