cmake_minimum_required(VERSION 3.24)
project(LibreWheel VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
option(LIBREWHEEL_BENCH_OUTPUT "Build the opt-in, torque-limited hardware validation path" OFF)
option(LIBREWHEEL_FULL_FORCE "Enable full motor output by default for hands-on testing" OFF)
if(LIBREWHEEL_FULL_FORCE AND NOT LIBREWHEEL_BENCH_OUTPUT)
    message(FATAL_ERROR "LIBREWHEEL_FULL_FORCE requires LIBREWHEEL_BENCH_OUTPUT")
endif()

set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(wheel_core STATIC src/effects.cpp src/settings.cpp src/protocol.cpp)
target_include_directories(wheel_core PUBLIC src)
target_compile_features(wheel_core PUBLIC cxx_std_20)

if(WIN32)
    add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN DIRECTINPUT_VERSION=0x0800
                            _WIN32_WINNT=0x0A00)
    add_library(wheel_hid STATIC src/windows/hid.cpp)
    target_link_libraries(wheel_hid PUBLIC wheel_core hid setupapi)
    add_library(LibreWheelFF SHARED src/windows/driver.cpp src/windows/driver.def)
    target_link_libraries(LibreWheelFF PRIVATE wheel_hid dxguid uuid)
    if(LIBREWHEEL_BENCH_OUTPUT)
        target_sources(LibreWheelFF PRIVATE src/windows/output_lease.cpp)
        target_compile_definitions(LibreWheelFF PRIVATE LIBREWHEEL_BENCH_OUTPUT)
        add_executable(LibreWheelOutput src/windows/broker.cpp)
        target_link_libraries(LibreWheelOutput PRIVATE wheel_hid)
        add_dependencies(LibreWheelFF LibreWheelOutput)
        if(LIBREWHEEL_FULL_FORCE)
            target_compile_definitions(LibreWheelFF PRIVATE LIBREWHEEL_FULL_FORCE)
            target_compile_definitions(LibreWheelOutput PRIVATE LIBREWHEEL_FULL_FORCE)
        endif()
    endif()
    add_executable(librewheel src/windows/cli.cpp)
    target_link_libraries(librewheel PRIVATE wheel_hid dinput8 dxguid uuid cfgmgr32)
endif()

if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE wheel_core)
    add_test(NAME core COMMAND core_tests)
    if(WIN32)
        add_executable(driver_tests tests/driver_tests.cpp src/windows/driver.cpp)
        target_link_libraries(driver_tests PRIVATE wheel_hid dxguid uuid)
        target_compile_definitions(driver_tests PRIVATE LIBREWHEEL_TEST_DRIVER)
        add_test(NAME driver COMMAND driver_tests $<TARGET_FILE:LibreWheelFF>)
        add_executable(output_tests tests/output_tests.cpp)
        target_link_libraries(output_tests PRIVATE wheel_core)
        add_test(NAME output COMMAND output_tests)
    endif()
endif()
